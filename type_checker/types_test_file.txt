let equal: type = fn(type, Equalable, Equalable) -> Logical;
// описание сигнатуры функции, первым параметром идёт type для избежания кофликтов имён в реализациях

let not_equal: type = fn(type, Equalable, Equalable) -> Logical; 
let Equalable: type = equal | not_equal; // интерфейсный тип

let less: type = fn(type, Comparable, Comparable) -> Logical; 
let greater: type = fn(type, Comparable, Comparable) -> Logical; 
let Comparable: type = less | greater;

let add: type = fn(type, Addable, Addable) -> Algebraic;
let Addable: type = add;

let sub: type = fn(type, Substractable, Substractable) -> Algebraic;
let Substractable: type = sub;

let mul: type = fn(type, Multipliable, Multipliable) -> Algebraic;
let Multipliable: type = mul;

let div: type = fn(type, Diviable, Diviable) -> Algebraic;
let Diviable: type = div;

let Algebraic: type = Addable | Substractable | Multipliable | Diviable;

let and: type = fn(type, Logical, Logical) -> Logical;
let or: type = fn(type, Logical, Logical) -> Logical;
let not: type = fn(type, Logical) -> Logical;
let Logical: type = and | or | not;

let int: type = type(Algebraic | Comparable | Equalable); // конкретная реализация типа

// реализация функции интерфейса через . для избежания конфликтов имён в интерфейсах, обращение также через .
let Algebraic.add: fn(type, int, int) -> int = fn(T, a, b) {
    return a + b;
}
